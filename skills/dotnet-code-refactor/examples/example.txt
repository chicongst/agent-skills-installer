### Characterization tests (written and green on the original code)
tests/Shop.Tests/Orders/OrderServiceCharacterizationTests.cs:
```csharp
public class OrderServiceCharacterizationTests
{
    private readonly TestDbFactory _db = new();
    private readonly IFulfillmentClient _fulfillment = Substitute.For<IFulfillmentClient>();

    private OrderService CreateSut(Dictionary<string, string?>? settings = null)
    {
        var config = new ConfigurationBuilder().AddInMemoryCollection(settings).Build();
        return new ServiceCollection().AddOrders(config).AddSingleton<IConfiguration>(config)
            .AddSingleton<IDbContextFactory<ShopDbContext>>(_db).AddSingleton(_fulfillment)
            .BuildServiceProvider().GetRequiredService<OrderService>();
    }

    private static PlaceOrderRequest Request(Guid customerId, params OrderLineRequest[] lines) =>
        new(customerId, [.. lines], ExpressShipping: false);

    [Fact]
    public async Task NoLines_ReturnsEmptyGuid_AndEnqueuesNothing()
    {
        Assert.Equal(Guid.Empty, await CreateSut().PlaceOrderAsync(Request(Seed.RegularId), default));
        Assert.Empty(_fulfillment.ReceivedCalls());
    }
    [Theory]
    [InlineData(" ", 1, "SKU is required. (Parameter 'request')")]
    [InlineData("A-1", 0, "Quantity must be positive for A-1. (Parameter 'request')")]
    public async Task InvalidLine_ThrowsSameArgumentException_FromBothEntryPoints(string sku, int qty, string message)
    {
        var sut = CreateSut();
        var request = Request(Seed.RegularId, new(sku, qty));
        var place = await Assert.ThrowsAsync<ArgumentException>(() => sut.PlaceOrderAsync(request, default)); // exact type
        var quote = await Assert.ThrowsAsync<ArgumentException>(() => sut.QuoteAsync(request, default));
        Assert.Equal(message, place.Message);
        Assert.Equal(message, quote.Message);
    }
    [Fact]
    public async Task UnknownCustomer_ThrowsKeyNotFound()
    {
        var missing = Guid.NewGuid();
        var ex = await Assert.ThrowsAsync<KeyNotFoundException>(() => CreateSut().PlaceOrderAsync(Request(missing, new("A-1", 1)), default));
        Assert.Equal($"Customer {missing} not found.", ex.Message);
    }
    [Fact]
    public async Task UnknownSku_ThrowsInvalidOperation_AndSavesNothing()
    {
        var ex = await Assert.ThrowsAsync<InvalidOperationException>(() => CreateSut().PlaceOrderAsync(Request(Seed.RegularId, new("X-9", 1)), default));
        Assert.Equal("Unknown SKU 'X-9'.", ex.Message);
        using var db = _db.CreateDbContext();
        Assert.Empty(db.Orders);
    }
    [Fact]
    public async Task VipDiscount_RoundsMidpointToEven() // 12.30 * 0.95 = 11.685 → 11.68 today
    {
        var id = await CreateSut().PlaceOrderAsync(Request(Seed.VipId, new("A-1", 1)), default);
        using var db = _db.CreateDbContext();
        Assert.Equal(11.68m, db.Orders.Single(o => o.Id == id).Total);
    }
    [Fact]
    public async Task NoOrdersSection_UsesUsd_AndPassesNullStandardQueue()
    {
        var id = await CreateSut().PlaceOrderAsync(Request(Seed.RegularId, new("A-1", 2)), default);
        await _fulfillment.Received(1).EnqueueAsync(Arg.Is<string>(q => q == null), id, Arg.Any<CancellationToken>());
        using var db = _db.CreateDbContext();
        Assert.Equal("USD", db.Orders.Single(o => o.Id == id).Currency);
    }
    [Fact]
    public async Task OrdersSection_IsUsed_ForCurrencyAndExpressQueue()
    {
        var sut = CreateSut(new() { ["Orders:DefaultCurrency"] = "EUR", ["Orders:ExpressQueue"] = "exp-eu" });
        var id = await sut.PlaceOrderAsync(Request(Seed.RegularId, new("A-1", 1)) with { ExpressShipping = true }, default);
        await _fulfillment.Received(1).EnqueueAsync("exp-eu", id, Arg.Any<CancellationToken>());
        using var db = _db.CreateDbContext();
        Assert.Equal("EUR", db.Orders.Single(o => o.Id == id).Currency);
    }
}
```

## Deliberately not changed
- VIP discount (lines 39-41): DANGEROUS (money); the user said to leave it out.
- `QuoteAsync` keeps `async`/`await`. Returning the query task without `await` would dispose `db` while the query may still run: broken on Npgsql, yet green on the SQLite fixture.
- One query per line in the product loop (line 29): a performance change on a hot query → `performance-review`.
- CS8604 on the standard queue: fixing it means deciding what a missing queue should do (see below).
## Behavior changes recommended (not applied)
- 🟠 MAJOR VIP discount uses banker's rounding — `OrderService.cs:41` — `Math.Round(decimal, 2)` defaults to `MidpointRounding.ToEven`, so 11.685 → 11.68, not 11.69 → confirm the rule with finance; if it should be `MidpointRounding.AwayFromZero`, change it and the characterization test together.
- 🟠 MAJOR Empty order silently returns `Guid.Empty` — `OrderService.cs:13,55` — the controller gets a success with an empty id → throw `ArgumentException` (or return 400) in its own change, updating `NoLines_ReturnsEmptyGuid_AndEnqueuesNothing`.
- 🟡 MINOR Missing `Orders:StandardQueue` passes `null` to fulfillment — `OrderService.cs:47` — fails later, far from the cause → give it a default, or fail fast with `AddOptions<OrderOptions>().Bind(...).Validate(o => o.StandardQueue is not null).ValidateOnStart()`.
