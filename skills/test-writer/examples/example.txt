## User Input
Write tests for this discount calculation function (it lives in `pricing.py`):

```python
def calculate_discount(price: float, user_tier: str, coupon_code: str | None = None) -> float:
    """Returns final price after discounts."""
    if price <= 0:
        raise ValueError("Price must be positive")

    discount = 0.0
    if user_tier == "premium":
        discount += 0.10
    elif user_tier == "enterprise":
        discount += 0.20

    if coupon_code == "SAVE10":
        discount += 0.10
    elif coupon_code == "SAVE20":
        discount += 0.20

    discount = min(discount, 0.50)  # max 50% discount
    return round(price * (1 - discount), 2)
```

## Test Code

```python
import math

import pytest

from pricing import calculate_discount


@pytest.mark.parametrize(
    ("tier", "coupon", "expected"),
    [
        ("free", None, 100.00),
        ("premium", None, 90.00),
        ("enterprise", None, 80.00),
        ("free", "SAVE10", 90.00),
        ("free", "SAVE20", 80.00),
        ("premium", "SAVE10", 80.00),
        ("premium", "SAVE20", 70.00),
        ("enterprise", "SAVE10", 70.00),
        ("enterprise", "SAVE20", 60.00),
    ],
)
def test_tier_and_coupon_discounts_stack_additively(tier, coupon, expected):
    assert calculate_discount(100.0, tier, coupon_code=coupon) == expected


def test_largest_possible_stack_is_40_percent_so_50_percent_cap_never_applies():
    # enterprise (20%) + SAVE20 (20%) is the maximum; the cap on line 17 is not reached (F1)
    assert calculate_discount(200.0, "enterprise", coupon_code="SAVE20") == 120.00


def test_unknown_tier_gets_no_discount():
    assert calculate_discount(100.0, "vip") == 100.00


def test_unknown_coupon_gets_no_discount():
    assert calculate_discount(100.0, "free", coupon_code="FAKE50") == 100.00


def test_tier_and_coupon_matching_is_case_sensitive():
    # Current behavior; whether "Premium" / "save10" should match is open question F3
    assert calculate_discount(100.0, "Premium", coupon_code="save10") == 100.00


def test_result_is_rounded_to_cents():
    # 99.99 * 0.90 = 89.991 -> 89.99
    assert calculate_discount(99.99, "premium") == 89.99


def test_smallest_cent_price_with_max_discount_rounds_up_to_one_cent():
    # 0.01 * 0.60 = 0.006 -> 0.01
    assert calculate_discount(0.01, "enterprise", coupon_code="SAVE20") == 0.01


@pytest.mark.parametrize("price", [0, -0.01, -50.0])
def test_non_positive_price_raises_value_error(price):
    with pytest.raises(ValueError, match="Price must be positive"):
        calculate_discount(price, "free")


@pytest.mark.xfail(strict=True, reason="F2: NaN passes the `price <= 0` guard and is returned as nan")
def test_nan_price_is_rejected():
    with pytest.raises(ValueError, match="Price must be positive"):
        calculate_discount(math.nan, "free")
```

## Coverage Gaps
- The 50% cap on line 17 — unreachable, so it cannot be tested without changing the code (F1).
- Sub-cent prices: `calculate_discount(0.004, "free")` returns `0.0`, so a positive price becomes free — not tested because callers presumably pass whole cents (unconfirmed assumption); ask whether prices below 0.01 should be rejected.
- Non-string `user_tier` / `coupon_code` (e.g. `None` tier) — type hints exclude them; low risk unless callers are untyped.
- Money is `float`; switching to `Decimal` would change the rounding behavior and needs its own tests.
